class CloneCutter {
    constructor() {
        this.folders = [];
        this.duplicates = [];
        this.trash = [];
        this.worker = null;
        this.init();
    }

    init() {
        this.bindEvents();
        this.checkPwaSupport();
    }

    bindEvents() {
        document.getElementById('selectFolders').addEventListener('click', () => this.selectFolders());
        document.getElementById('startScan').addEventListener('click', () => this.startScan());
        document.getElementById('selectAll').addEventListener('click', () => this.selectAll());
        document.getElementById('clearAll').addEventListener('click', () => this.clearAll());
        document.getElementById('removeSelected').addEventListener('click', () => this.removeSelected());
        document.getElementById('emptyTrash').addEventListener('click', () => this.emptyTrash());
        document.getElementById('restoreAll').addEventListener('click', () => this.restoreAll());

        document.querySelector('.file-types').addEventListener('change', (e) => {
            if (e.target.value === 'all') {
                this.toggleAllFileTypes(e.target.checked);
            }
            this.updateStartButton();
        });
    }

    async selectFolders() {
        try {
            if ('showDirectoryPicker' in window) {
                const dirHandle = await window.showDirectoryPicker();
                const folderInfo = await this.getFolderInfo(dirHandle);
                this.addFolder(folderInfo);
            } else {
                alert('Directory picker not supported. Please use Chrome/Edge with File System Access API.');
            }
        } catch (err) {
            if (err.name !== 'AbortError') {
                console.error('Folder selection error:', err);
            }
        }
    }

    async getFolderInfo(dirHandle) {
        const folderName = dirHandle.name;
        const fileCount = await this.countFiles(dirHandle);
        return { handle: dirHandle, name: folderName, fileCount, path: folderName };
    }

    async countFiles(dirHandle, count = 0) {
        for await (const entry of dirHandle.values()) {
            if (entry.kind === 'file') {
                count++;
            } else if (entry.kind === 'directory') {
                count += await this.countFiles(entry);
            }
        }
        return count;
    }

    addFolder(folderInfo) {
        const exists = this.folders.some(f => f.name === folderInfo.name);
        if (!exists) {
            this.folders.push(folderInfo);
            this.renderFolders();
            this.updateStartButton();
        }
    }

    renderFolders() {
        const container = document.getElementById('folderList');
        container.innerHTML = this.folders.map(folder => `
            <div class="folder-item">
                📁 ${folder.name} (${folder.fileCount.toLocaleString()} files)
                <button onclick="cloneCutter.removeFolder('${folder.name}')">Remove</button>
            </div>
        `).join('');
    }

    removeFolder(folderName) {
        this.folders = this.folders.filter(f => f.name !== folderName);
        this.renderFolders();
        this.updateStartButton();
    }

    toggleAllFileTypes(checked) {
        document.querySelectorAll('.file-types input:not([value="all"])').forEach(cb => {
            cb.checked = checked;
        });
    }

    updateStartButton() {
        const hasFolders = this.folders.length > 0;
        document.getElementById('startScan').disabled = !hasFolders;
    }

    async startScan() {
        if (this.folders.length === 0) return;

        this.showProgress();
        this.createWorker();

        const fileTypes = Array.from(document.querySelectorAll('.file-types input:checked'))
            .filter(cb => cb.value !== 'all')
            .map(cb => cb.value);

        const files = [];
        for (const folder of this.folders) {
            const folderFiles = await this.readFolderFiles(folder.handle, '');
            files.push(...folderFiles);
        }

        this.worker.postMessage({
            action: 'scan',
            files: files.map(f => ({ path: f.path, handle: f.handle })),
            fileTypes
        });
    }

    async readFolderFiles(dirHandle, pathPrefix) {
        const files = [];
        for await (const entry of dirHandle.values()) {
            const fullPath = pathPrefix ? `${pathPrefix}/${entry.name}` : entry.name;
            
            if (entry.kind === 'file') {
                files.push({ path: fullPath, handle: entry });
            } else if (entry.kind === 'directory') {
                files.push(...await this.readFolderFiles(entry, fullPath));
            }
        }
        return files;
    }

    createWorker() {
        if (this.worker) {
            this.worker.terminate();
        }

        const workerCode = this.worker.toString();
        const blob = new Blob([`(${workerCode})()`], { type: 'application/javascript' });
        this.worker = new Worker(URL.createObjectURL(blob));

        this.worker.onmessage = (e) => this.handleWorkerMessage(e.data);
        this.worker.onerror = (e) => {
            console.error('Worker error:', e);
            this.showError('Scan failed. Please try again.');
        };
    }

    handleWorkerMessage(data) {
        switch (data.type) {
            case 'progress':
                this.updateProgress(data.progress, data.message);
                break;
            case 'complete':
                this.duplicates = data.duplicates;
                this.renderResults();
                this.hideProgress();
                break;
            case 'error':
                this.showError(data.message);
                break;
        }
    }

    showProgress() {
        document.querySelector('.upload-section').classList.add('hidden');
        document.getElementById('progressSection').classList.remove('hidden');
        document.getElementById('resultsSection').classList.add('hidden');
    }

    hideProgress() {
        document.getElementById('progressSection').classList.add('hidden');
        document.getElementById('resultsSection').classList.remove('hidden');
    }

    updateProgress(progress, message) {
        document.getElementById('progressFill').style.width = `${progress}%`;
        document.getElementById('progressText').textContent = message;
    }

    renderResults() {
        this.updateStats();
        const container = document.getElementById('duplicatesList');
        container.innerHTML = this.duplicates.map(group => this.renderDuplicateGroup(group)).join('');

        // Bind group toggle events
        document.querySelectorAll('.group-header').forEach(header => {
            header.addEventListener('click', () => {
                const group = header.parentElement;
                group.classList.toggle('expanded');
                const toggle = header.querySelector('.group-toggle');
                toggle.textContent = group.classList.contains('expanded') ? '▼' : '▶';
            });
        });

        // Bind file selection events
        document.querySelectorAll('.file-checkbox').forEach(cb => {
            cb.addEventListener('change', () => this.updateRemoveButton());
        });
    }

    renderDuplicateGroup(group) {
        const totalSize = group.files.reduce((sum, f) => sum + f.size, 0);
        const spaceSaved = (group.files.length - 1) * group.files[0].size;
        
        return `
            <div class="duplicate-group" data-group-id="${group.id}">
                <div class="group-header">
                    <div class="group-info">
                        <h4>${group.hash.substring(0, 16)}...</h4>
                        <div class="group-stats">
                            ${group.files.length} identical files • 
                            ${this.formatBytes(totalSize)} total • 
                            Save
